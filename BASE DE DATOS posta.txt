/*CREATE DATABASE posta*/
USE posta;
/*creacion de tablas*/

CREATE TABLE administrador
(
idadmin INT NOT NULL PRIMARY KEY,
apellido VARCHAR(30),
nombres VARCHAR(30),
correo VARCHAR (50),
clave CHAR(5)
);


CREATE TABLE paciente (
dni CHAR(8) NOT NULL PRIMARY KEY,
apellidos VARCHAR(20),
nombres VARCHAR(20),
telefono CHAR(10)
);

CREATE TABLE cita (
numero INT NOT NULL PRIMARY KEY,
fecha DATE,
turno VARCHAR(20),
hora VARCHAR(20),
dni CHAR(8),
idespe CHAR(5)
);

CREATE TABLE especialidad (
idespe CHAR(5) NOT NULL PRIMARY KEY,
nombre_espe VARCHAR(50)
);

/*establecer las relaciones*/

/*relacion entre cita y paciente*/
ALTER TABLE cita ADD FOREIGN KEY(dni) REFERENCES paciente(dni);

/*relacion entre cita y especialidad */
ALTER TABLE cita ADD FOREIGN KEY(idespe) REFERENCES paciente(idespe);
